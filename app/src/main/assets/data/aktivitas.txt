# ==========================================================
#  DAFTAR AKTIVITAS WAKTU BEBAS
#  Bisa ditambah/diubah dari HP. Panduan: docs/DATA.md
#
#  [id_aktivitas]
#  nama   = nama yang tampil
#  ikon   = emoji
#  tempat = desa, kota, atau semua
#  waktu  = pagi, siang, sore, malam (pisahkan koma), atau semua
#           (Pagi 04:00–10:59 · Siang 11:00–14:59 · Sore 15:00–17:59 · Malam 18:00–03:59)
#  durasi = lama kegiatan: 45 menit / 2 jam / 1,5 jam (bawaan 1 jam)
#  mulai  = (opsional) jam boleh MULAI, contoh 04:00-06:00 (melaut subuh) atau 19:00-02:00 (lewat tengah malam).
#           Di luar jam itu aktivitas tetap tampil REDUP dengan keterangan, mis. "Sudah terlalu siang untuk
#           melaut bersama Ayah (paling lambat 06:00)". Tanpa "mulai", yang dipakai "waktu".
#  Aktivitas hanya tampil di "tempat"-nya: aktivitas desa tidak muncul saat pemain di kota, dan sebaliknya.
#  efek   = energi, kenyang, hidrasi, kewarasan, kesehatan, uang (+/-),
#           atau variabel lain (mis. akademik +2)
#  syarat = (opsional) contoh: jalur != kerja   atau   usia >= 18
#  teks   = narasi yang tampil setelah aktivitas dipilih
#  batas  = (opsional) paling banyak berapa kali per hari, contoh 1
#  jeda   = (opsional) jarak minimal sebelum boleh diulang, contoh 4 jam (makan) — dihitung dari jam mulai
#  teks_ulang = (opsional) narasi bila dilakukan lagi di hari yang sama (mis. Ibu menegur)
#  Yang sudah mencapai batas/jeda tampil REDUP: "Sudah dilakukan hari ini — besok lagi" /
#  "Baru saja dilakukan — bisa lagi pukul 18:45".
# ==========================================================
