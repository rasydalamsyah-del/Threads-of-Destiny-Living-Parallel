# ==========================================================
#  SEKOLAH & PEKERJAAN
#  Panduan lengkap: docs/DATA.md
#
#  [id]
#  nama        = nama yang tampil
#  ikon        = emoji
#  jenis       = sekolah / tetap / paruh_waktu / harian
#                sekolah & tetap WAJIB hadir sesuai jadwal (bolos/absen tercatat:
#                3x → surat peringatan, 6x → dikeluarkan/dipecat)
#                paruh_waktu & harian bebas datang kapan saja sesuai jadwal
#  tempat      = desa / kota
#  hari        = senin-sabtu  /  senin, rabu, jumat  /  setiap hari
#  waktu       = pagi, siang, sore, malam   (bagian hari; dipakai bila "jam" kosong)
#  jam         = jam masuk–pulang, contoh 07:00-14:00. Bisa berangkat 1 jam sebelum masuk;
#                lewat 15 menit = terlambat (kinerja turun); tidak datang sampai 1 jam sebelum pulang = bolos/absen
#  gaji        = paruh_waktu & harian: per hari kerja (satu shift) · tetap: per 30 hari
#  efek        = efek setiap kali hadir (energi, kenyang, kewarasan, akademik, dll.)
#  usia_min    = usia minimal
#  syarat      = (opsional) syarat lain, contoh: jalur != kerja
#  cara        = langsung (bisa dicari di "Cari kerja") / lamar / cerita (hanya lewat naskah)
#  sulit       = 0-100, makin tinggi makin sulit diterima
#  orang_dalam = id karakter yang bisa merekomendasikan (Cinta & Kepercayaan ≥ 60)
#  berkas      = (untuk cara = lamar) dokumen yang wajib ada, dari dokumen.txt
#  tes         = ya / tidak  (untuk cara = lamar: ada tes tertulis sebelum wawancara)
#  ujian_tiap  = (sekolah) ujian setiap N hari. Setiap 365 hari: naik kelas / tinggal kelas / lulus
#  keterangan  = penjelasan singkat
#  teks        = narasi setiap kali hadir
# ==========================================================
