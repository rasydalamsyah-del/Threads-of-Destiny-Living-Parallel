# ==========================================================
#  PERTANYAAN WAWANCARA KERJA
#  Saat wawancara, 3 pertanyaan dipilih acak.
#  [id]
#  tanya = pertanyaan pewawancara
#  a / b / c / d = jawaban | skor | bohong (opsional)
#     skor ditambahkan ke nilai wawancara
#     "bohong" = jawaban berbohong: skor tinggi, tapi bisa ketahuan (skor −30)
# ==========================================================
