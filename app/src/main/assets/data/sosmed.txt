#  Panduan lengkap: docs/DATA.md
#
#  --- Unggahan teman (beranda) ---
#  [id]
#  penulis = id karakter (hanya tampil bila pemain sudah kenal)
#  isi     = isi unggahan ({nama} = nama pemain)
#  hari    = mulai tampil pada hari ke-sekian (0 = sejak awal hidup dimulai)
#  syarat  = (opsional) contoh: jalur == kerja   ·   lokasi == kota
#  Pemain bisa menyukai unggahan sekali → Cinta penulis +1.
#
#  --- Hasil pencarian sahabat ---
#  [id]
#  jenis    = cari
#  butuh    = ditemukan pada pencarian ke-sekian (pemain hanya bisa mencari 1x sehari)
#  syarat   = (opsional)
#  isi      = apa yang ditemukan ({nama_sahabat} = nama sahabat)
#  catatan  = disimpan ke aplikasi Catatan
#  petunjuk = tambahan petunjuk_sahabat
# ==========================================================
