# ==========================================================
#  DAFTAR KARAKTER
#  Panduan lengkap: docs/DATA.md
#
#  [id]                 → id unik (dipakai di naskah: @tampil bayu senyum)
#  nama        = nama yang tampil (dan dipakai sebagai nama pembicara di naskah)
#  kelamin     = laki-laki / perempuan / lawan (= lawan jenis pemain)
#  nama_l / nama_p = (opsional) nama bila karakter laki-laki / perempuan
#  usia        = usia saat cerita dimulai (ikut bertambah bersama usia pemain)
#  skala       = (opsional) tinggi gambar di layar dibanding orang dewasa (1.0). Anak 11 tahun ± 0.78, remaja ± 0.95
#  peran       = keterangan singkat
#  sifat       = keterangan sifat (untuk penulis)
#  pandangan   = setia / terbuka   (sikap bila pasangannya punya pasangan lain)
#  bisa_dipacari = ya / tidak      (romansa hanya bila pemain & karakter sudah 18+)
#  tempat      = di mana bisa ditemui: desa / kota
#  waktu       = kapan bisa ditemui: pagi, siang, sore, malam
#  ditemui     = ya / tidak / janji (janji = hanya bisa ditemui lewat janji di Chat → "Ajak ketemu")
#  jadwal      = (untuk karakter TANPA kolom lokasi) kapan bisa ditemui langsung, contoh: senin-sabtu 13:00-20:00
#                (tanpa jadwal: mengikuti "waktu"). "waktu" juga dipakai untuk pilihan jam saat diajak janjian
#  cinta_awal, percaya_awal = nilai awal 0-100
#  suka_topik, benci_topik   = id dari topik.txt
#  suka_barang, benci_barang = id dari barang.txt
#  sapa        = kalimat saat ditemui
#  respon_suka / respon_biasa / respon_benci = jawaban saat diajak ngobrol
#  hadiah_suka / hadiah_biasa / hadiah_benci = jawaban saat diberi hadiah
#  larangan    = kalimat saat ia dilarang bertemu oleh orang dekatnya
#  lokasi      = di mana ia berada & kapan (mis. setiap hari 12:00-13:00 ruang_makan, ...) → menggantikan jadwal:
#                bisa ditemui tepat saat ia ada di salah satu lokasinya. Jadwal lengkap desa: docs/CERITA.md
#  ada_bila    = (opsional) syarat karakter masih ada di tempatnya, mis. sekar_pergi != 1
#  tempat_janji = lokasi tempat ia menunggu saat kalian janjian (untuk ditemui = janji)
#  (pesan chat yang ia kirim & pilihan balasannya ada di data/pesan.txt)
#  ajak_ya / ajak_tidak = jawaban saat diajak ketemu lewat chat
#  warna       = warna nama (#RRGGBB)
#
#  --- PROFIL (halaman ☰ → Karakter). Terbuka bertahap, lihat docs/DATA.md ---
#  hobi, kegemaran, pelajaran (anak sekolah) / keahlian (dewasa), cita_cita
#  kisah_1, kisah_2, kisah_3 = cerita hidupnya, makin dalam makin sulit dibuka
#  tahu_awal   = info yang sudah diketahui sejak kenal (mis. keluarga sendiri)
#  tampil_awal = ya → sudah tampil di daftar meski belum bertemu (mis. sahabat masa kecil)
#  daftar      = tidak → tidak tampil di daftar Karakter
# ==========================================================
