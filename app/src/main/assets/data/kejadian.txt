# ==========================================================
#  KEJADIAN ACAK & RINTANGAN
#  Setiap ganti hari, paling banyak SATU kejadian bisa terjadi.
#  Muncul saat waktu bebas berikutnya. Bisa juga dipanggil dari naskah: @kejadian id
#
#  [id]
#  judul    = judul singkat
#  ikon     = emoji
#  tempat   = desa / kota / semua
#  peluang  = persen kemungkinan per hari (0 = hanya lewat naskah)
#  syarat   = (opsional) contoh: uang >= 30000   ·   gabungkan dengan &&
#             variabel berguna: uang, lokasi, kos, musim (hujan/kemarau), bulan, usia,
#             kenal_<id>, cinta_<id>, jalur, kerja_<id>
#  sekali   = ya (hanya terjadi sekali seumur permainan)
#  teks     = cerita kejadian
#  efek     = efek langsung (bar, uang, uang -40%, cinta_ibu +5, percaya_bayu -5, variabel lain)
#  pilih_a / pilih_b / pilih_c = Teks tombol | efek | teks hasil
#  catatan  = (opsional) tambahkan catatan ke aplikasi Catatan
# ==========================================================
