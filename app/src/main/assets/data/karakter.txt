# ==========================================================
#  DAFTAR KARAKTER
#  Panduan lengkap: docs/DATA.md
#
#  [id]                 → id unik (dipakai di naskah: @tampil bayu senyum)
#  nama        = nama yang tampil (dan dipakai sebagai nama pembicara di naskah)
#  kelamin     = laki-laki / perempuan
#  usia        = usia saat cerita dimulai (ikut bertambah bersama usia pemain)
#  peran       = keterangan singkat
#  sifat       = keterangan sifat (untuk penulis)
#  pandangan   = setia / terbuka   (sikap bila pasangannya punya pasangan lain)
#  bisa_dipacari = ya / tidak      (romansa hanya bila pemain & karakter sudah 18+)
#  tempat      = di mana bisa ditemui: desa / kota
#  waktu       = kapan bisa ditemui: pagi, siang, sore, malam
#  ditemui     = ya / tidak        (muncul di daftar "Temui seseorang" saat waktu bebas)
#  cinta_awal, percaya_awal = nilai awal 0-100
#  suka_topik, benci_topik   = id dari topik.txt
#  suka_barang, benci_barang = id dari barang.txt
#  sapa        = kalimat saat ditemui
#  respon_suka / respon_biasa / respon_benci = jawaban saat diajak ngobrol
#  hadiah_suka / hadiah_biasa / hadiah_benci = jawaban saat diberi hadiah
#  larangan    = kalimat saat ia dilarang bertemu oleh orang dekatnya
#  warna       = warna nama (#RRGGBB)
# ==========================================================
